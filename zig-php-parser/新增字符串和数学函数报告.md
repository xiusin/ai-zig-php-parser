```zig
/// pow - 幂运算（优化版）
pub fn php_pow_func(base: Value, exponent: Value) !Value {
    if (base.isInt() and exponent.isInt()) {
        // 整数幂运算优化
        const b = base.asInt();
        const e = exponent.asInt();
        if (e >= 0 and e < 64) {
            var result: i64 = 1;
            var i: i64 = 0;
            while (i < e) : (i += 1) {
                result *= b;
            }
            return Value.initInt(result);
        }
    }
    return Value.initFloat(std.math.pow(f64, base.toFloat(), exponent.toFloat()));
}
```

### 4. Allocator 参数自动处理

实现了智能的 allocator 参数管理：

```zig
/// 检查函数是否需要 allocator 参数
fn functionNeedsAllocator(self: *const Self, func_name: []const u8) bool {
    const needs_allocator = [_][]const u8{
        "strtoupper", "strtolower", "trim", "ltrim", "rtrim",
        "str_replace", "str_repeat", "str_pad", "strrev",
        "ucfirst", "lcfirst", "ucwords",
        "explode", "implode", "str_split",
        "strcasecmp", "substr", "strval",
    };
    
    for (needs_allocator) |name| {
        if (std.mem.eql(u8, func_name, name)) return true;
    }
    
    return false;
}

// 在函数调用生成时自动添加 allocator 参数
if (self.functionNeedsAllocator(op.func_name)) {
    if (args_buf.items.len > 0) {
        try writer.print("reg_{d} = try runtime.{s}({s}, runtime.runtime_allocator);\n", 
            .{reg.id, runtime_name, args_buf.items});
    } else {
        try writer.print("reg_{d} = try runtime.{s}(runtime.runtime_allocator);\n", 
            .{reg.id, runtime_name});
    }
}
```
