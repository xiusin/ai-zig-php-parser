## 实现架构

### 数据流

```
PHP源码
  $obj = new class implements Logger {
      private $prefix = "[LOG]";
      public function log($msg) { ... }
  };
    ↓
AST解析
  anonymous_class节点
    ↓
IR生成（ir_generator.zig）
  1. 生成唯一类名: anonymous_class_{id}
  2. internString确保持久化
  3. 创建TypeDef
  4. 注册到module.types
  5. 进入类作用域
  6. 处理成员:
     - method_decl → generateMethodDecl
     - property_decl → generatePropertyDecl
  7. 离开作用域
  8. 生成new_object指令
    ↓
IR表示
  TypeDef { name: "anonymous_class_0", properties: [...], methods: [...] }
  new_object { class_name: "anonymous_class_0", args: [...] }
    ↓
Codegen（native_linker.zig）
  1. 生成ClassMeta.init
  2. 添加属性（addProperty）
  3. 添加方法（addMethod）
  4. 注册类（registerClass）
  5. 生成对象创建调用
    ↓
生成代码
  const meta = try ClassMeta.init(allocator, "anonymous_class_0");
  try meta.addProperty(.{ .name = "prefix", .default_value = ... });
  try meta.addMethod(.{ .name = "log", .func = @"anonymous_class_0::log" });
  try registerClass(meta);
  reg_0 = try php_object_new_with_constructor("anonymous_class_0", &[_]Value{}, allocator);
    ↓
Runtime
  标准对象创建流程
```

## 关键代码

### IR生成（ir_generator.zig）

```zig
fn generateAnonymousClass(self: *Self, node: *const Node) !Register {
    const anon_data = node.data.anonymous_class;
    
    // 1. 生成唯一类名
    const anon_class_name_temp = try std.fmt.allocPrint(
        self.allocator,
        "anonymous_class_{d}",
        .{self.module.?.next_anon_class_id}
    );
    defer self.allocator.free(anon_class_name_temp);
    self.module.?.next_anon_class_id += 1;
    
    // 2. Intern字符串确保持久化
    const anon_class_name_sid = try self.module.?.internString(anon_class_name_temp);
    const anon_class_name = self.module.?.getString(anon_class_name_sid) orelse return error.StringNotFound;
    
    // 3. 创建TypeDef
    const type_def = try self.allocator.create(TypeDef);
    type_def.* = .{
        .name = anon_class_name,
        .kind = .class,
        .parent = if (anon_data.extends) |ext_idx| ... else null,
        .interfaces = &.{},
        .traits = &.{},
        .properties = &.{},
        .methods = &.{},
        .constants = &.{},
        .location = self.current_location,
    };
    
    // 4. 注册类
    if (self.module) |module| {
        try module.addTypeDef(type_def);
    }
    try self.symbol_table.defineClass(anon_class_name, type_def.parent, &.{}, self.current_location);
    
    // 5. 进入类作用域
    _ = try self.symbol_table.enterScope(.class, anon_class_name);
    defer self.symbol_table.leaveScope();
    
    // 6. 处理成员
    for (anon_data.members) |member_idx| {
        const member = self.getNode(member_idx) orelse continue;
        switch (member.tag) {
            .method_decl => try self.generateMethodDecl(member, anon_class_name),
            .property_decl => try self.generatePropertyDecl(member, anon_class_name),
            else => {},
        }
    }
    
    // 7. 生成构造函数参数
    var ctor_args = std.ArrayListUnmanaged(Register){};
    defer ctor_args.deinit(self.allocator);
    
    for (anon_data.args) |arg_idx| {
        const arg_reg = try self.generateExpression(arg_idx);
        try ctor_args.append(self.allocator, arg_reg);
    }
    
    // 8. 创建对象
    const args_slice = try self.allocator.dupe(Register, ctor_args.items);
    return self.emitWithResult(.{ .new_object = .{
        .class_name = anon_class_name,
        .args = args_slice,
    } }, .{ .php_object = anon_class_name });
}
```

### 属性注册（ir_generator.zig）

```zig
fn generatePropertyDecl(self: *Self, node: *const Node, class_name: []const u8) !void {
    const prop_data = node.data.property_decl;
    const prop_name = self.getString(prop_data.name);
    
    // 注册到symbol_table
    try self.symbol_table.defineVariable(prop_name, prop_type, self.current_location);
    
    // 添加到TypeDef
    if (self.module) |module| {
        for (module.types.items) |type_def_ptr| {
            if (std.mem.eql(u8, type_def_ptr.name, class_name)) {
                const prop_def = TypeDef.Property{
                    .name = prop_name,
                    .visibility = ...,
                    .is_static = ...,
                    .type_ = ...,
                    .default_value = if (prop_data.default_value) |val_idx| blk: {
                        const val_reg = try self.generateExpression(val_idx);
                        const block = self.current_block orelse break :blk null;
                        const inst = block.instructions.items[block.instructions.items.len - 1];
                        break :blk inst;
                    } else null,
                };
                
                // 扩展properties数组
                const new_props = try self.allocator.alloc(TypeDef.Property, type_def_ptr.properties.len + 1);
                @memcpy(new_props[0..type_def_ptr.properties.len], type_def_ptr.properties);
                new_props[type_def_ptr.properties.len] = prop_def;
                if (type_def_ptr.properties.len > 0) {
                    self.allocator.free(type_def_ptr.properties);
                }
                type_def_ptr.properties = new_props;
                break;
            }
        }
    }
}
```

### Codegen（native_linker.zig）

**类注册**:
```zig
// 遍历所有TypeDef
for (ir_module.types.items) |type_def_ptr| {
    const td = type_def_ptr.*;
    const cname = td.name;
    
    // 创建ClassMeta
    try writer.print("const {s}_meta = try runtime.ClassMeta.init(allocator, \"{s}\");\n", 
        .{ cname, cname });
    
    // 添加属性
    for (td.properties) |prop| {
        try writer.print("try {s}_meta.addProperty(.{{ .name = \"{s}\", .default_value = ", 
            .{ cname, prop.name });
        if (prop.default_value) |dv| {
            switch (dv.op) {
                .const_int => |v| try writer.print("runtime.Value.initInt({d})", .{v}),
                .const_string => |sid| try writer.print("runtime.Value.initString(...)", .{sid}),
                // ...
            }
        } else {
            try writer.writeAll("runtime.Value.initNull()");
        }
        try writer.print(", .is_static = {}, .is_public = {} }});\n", 
            .{ prop.is_static, is_public });
    }
    
    // 添加方法
    for (td.methods) |method| {
        try writer.print("try {s}_meta.addMethod(.{{ .name = \"{s}\", .func = @\"{s}::{s}\" }});\n",
            .{ cname, method.name, cname, method.name });
    }
    
    // 注册类
    try writer.print("try runtime.registerClass({s}_meta);\n", .{cname});
}
```

## 关键技术决策

### 1. 类名生成：全局计数器 vs UUID

**选择**: 全局计数器

**理由**:
- 简单高效
- 可预测（便于调试）
- 唯一性保证

**实现**:
```zig
pub const Module = struct {
    next_anon_class_id: u32 = 0,
    // ...
};
```

### 2. 字符串生命周期：栈 vs 堆 vs intern

**选择**: intern（模块级字符串池）

**理由**:
- TypeDef.name需要持久化
- 栈字符串会被释放（dangling pointer）
- 堆字符串需要手动管理
- intern自动管理生命周期

**实现**:
```zig
const name_temp = try std.fmt.allocPrint(...);  // 临时
defer self.allocator.free(name_temp);
const name_sid = try module.internString(name_temp);  // 持久化
const name = module.getString(name_sid);  // 安全引用
```

### 3. args生命周期：栈 vs 堆

**选择**: 堆（allocator.dupe）

**理由**:
- IR指令需要持久化args
- 栈数组会被释放
- dupe确保独立生命周期

**实现**:
```zig
// WRONG
const args = &[_]Register{reg1, reg2};  // 栈
.args = args  // Dangling!

// CORRECT
const args = try allocator.dupe(Register, &[_]Register{reg1, reg2});  // 堆
.args = args  // 安全
```

## 技术难点

### 1. 字符串生命周期

**问题**: 类名字符串何时释放？

**错误方案**:
```zig
const name = try std.fmt.allocPrint(...);
defer allocator.free(name);  // 立即释放
type_def.name = name;  // Dangling pointer!
```

**正确方案**:
```zig
const name_temp = try std.fmt.allocPrint(...);
defer allocator.free(name_temp);
const name_sid = try module.internString(name_temp);  // 持久化
const name = module.getString(name_sid);  // 安全
type_def.name = name;
```

**关键**: internString将字符串添加到模块级字符串池

### 2. args生命周期

**问题**: new_object的args何时释放？

**错误方案**:
```zig
const args = &[_]Register{reg1, reg2};  // 栈数组
.args = args  // 栈被释放后dangling!
```

**正确方案**:
```zig
const args = try allocator.dupe(Register, &[_]Register{reg1, reg2});  // 堆
.args = args  // 由IR.deinit释放
```

**关键**: 所有IR指令的slice必须堆分配

### 3. 属性默认值

**问题**: 属性默认值如何传递到runtime？

**方案**: 保存Instruction指针
```zig
.default_value = if (prop_data.default_value) |val_idx| blk: {
    const val_reg = try self.generateExpression(val_idx);
    const block = self.current_block orelse break :blk null;
    const inst = block.instructions.items[block.instructions.items.len - 1];
    break :blk inst;  // 保存指令指针
} else null,
```

**Codegen时**:
```zig
if (prop.default_value) |dv| {
    switch (dv.op) {
        .const_int => |v| try writer.print("Value.initInt({d})", .{v}),
        .const_string => |sid| try writer.print("Value.initString(...)", .{sid}),
        // ...
    }
}
```
