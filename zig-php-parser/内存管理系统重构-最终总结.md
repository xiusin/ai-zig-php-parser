## 💡 完整解决方案

### 方案A: 替换HashMap为ArrayList (推荐)
**优点**:
- 简单，不容易出错
- 避免HashMap的grow问题
- 性能影响小（变量表通常<100个）

**实现**:
```zig
// 1. 定义VarRegister
const VarRegister = struct {
    name: []const u8,
    reg: IR.Register,
};

// 2. 替换HashMap
var_registers: std.ArrayListUnmanaged(VarRegister),
var_usage: std.ArrayListUnmanaged(VarUsage),
ref_vars: std.ArrayListUnmanaged([]const u8),

// 3. 实现辅助方法
fn getVarRegister(self: *const Self, name: []const u8) ?Register {
    for (self.var_registers.items) |vr| {
        if (std.mem.eql(u8, vr.name, name)) return vr.reg;
    }
    return null;
}

fn putVarRegister(self: *Self, name: []const u8, reg: Register) !void {
    for (self.var_registers.items) |*vr| {
        if (std.mem.eql(u8, vr.name, name)) {
            vr.reg = reg;
            return;
        }
    }
    try self.var_registers.append(self.allocator, .{ .name = name, .reg = reg });
}
```

**工作量**: 2-3小时
**预期通过率**: 85%+
