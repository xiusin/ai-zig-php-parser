## 一、问题分析

### 1.1 泄漏根本原因

```zig
// 每次循环迭代:
// 1. 创建临时字符串 "x"
reg_18 = runtime.Value.initString(try runtime.PHPString.init(runtime.runtime_allocator, "x"));

// 2. 执行字符串拼接
reg_19 = try runtime.php_concat(var_result, reg_18, runtime.runtime_allocator);

// 3. 释放旧的 var_result ✅ (已修复)
var_result.release(runtime.runtime_allocator);

// 4. 赋值新值
var_result = reg_19;

// ❌ 问题: reg_18 (临时字符串"x") 从未被释放!
```

### 1.2 泄漏模式

- **泄漏对象**: 循环中创建的临时字符串常量
- **泄漏数量**: 与循环次数相同（500次循环 = 500个泄漏）
- **泄漏位置**: `runtime_lib.zig:56:48` (PHPString.init)

### 2.1 寄存器生命周期分析

#### 新增数据结构

```zig
/// 寄存器生命周期信息
const RegLifetime = struct {
    def_block: usize,        // 定义所在的块索引
    def_inst: usize,         // 定义所在的指令索引
    last_use_block: usize,   // 最后使用所在的块索引
    last_use_inst: usize,    // 最后使用所在的指令索引
    use_count: usize,        // 使用次数
    is_temp: bool,           // 是否为临时寄存器（只使用一次）
    needs_release: bool,     // 是否需要释放（字符串、数组等）
};
```

#### 生命周期分析流程

```zig
// 第一遍：收集所有寄存器定义
for (func.blocks.items, 0..) |block, block_idx| {
    for (block.instructions.items, 0..) |inst, inst_idx| {
        if (inst.result) |reg| {
            // 判断是否需要释放
            const needs_release = switch (inst.op) {
                .const_string, .concat, .interpolate, .array_new => true,
                else => false,
            };
            
            // 初始化生命周期信息
            try reg_lifetime.put(reg.id, .{
                .def_block = block_idx,
                .def_inst = inst_idx,
                .last_use_block = block_idx,
                .last_use_inst = inst_idx,
                .use_count = 0,
                .is_temp = false,
                .needs_release = needs_release,
            });
        }
    }
}

// 第二遍：记录所有寄存器使用
for (func.blocks.items, 0..) |block, block_idx| {
    for (block.instructions.items, 0..) |inst, inst_idx| {
        try self.recordRegisterUsesForLifetime(inst, block_idx, inst_idx, &reg_lifetime);
    }
}

// 第三遍：标记临时寄存器
var lifetime_iter = reg_lifetime.iterator();
while (lifetime_iter.next()) |entry| {
    const lifetime = entry.value_ptr;
    lifetime.is_temp = (lifetime.use_count == 1);
}
```

### 2.2 新增函数

#### recordRegisterUsesForLifetime

```zig
/// 记录指令中使用的所有寄存器（用于完整生命周期分析）
fn recordRegisterUsesForLifetime(
    self: *Self,
    inst: *const IR.Instruction,
    block_idx: usize,
    inst_idx: usize,
    reg_lifetime: *std.AutoHashMap(usize, RegLifetime),
) !void {
    // 辅助函数：更新寄存器使用信息
    const updateUse = struct {
        fn call(
            lifetime_map: *std.AutoHashMap(usize, RegLifetime),
            reg_id: usize,
            blk_idx: usize,
            ins_idx: usize,
        ) !void {
            if (lifetime_map.getPtr(reg_id)) |lifetime| {
                lifetime.last_use_block = blk_idx;
                lifetime.last_use_inst = ins_idx;
                lifetime.use_count += 1;
            }
        }
    }.call;
    
    // 根据指令类型记录使用的寄存器
    switch (inst.op) {
        .add, .sub, .mul, .div, .mod, .pow,
        .eq, .ne, .lt, .le, .gt, .ge,
        .identical, .not_identical,
        .and_, .or_, .concat => |op| {
            try updateUse(reg_lifetime, op.lhs.id, block_idx, inst_idx);
            try updateUse(reg_lifetime, op.rhs.id, block_idx, inst_idx);
        },
        // ... 其他指令类型
        else => {},
    }
}
```
