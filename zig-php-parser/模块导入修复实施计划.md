### 3. 批量更新导入语句

可以使用 Python 脚本批量替换：

```python
#!/usr/bin/env python3
import re
import os

def fix_imports(file_path):
    with open(file_path, 'r') as f:
        content = f.read()
    
    # 替换编译器导入
    content = re.sub(
        r'const (\w+) = @import\("\.\.\/compiler\/(\w+)\.zig"\);',
        r'const compiler = @import("compiler");\nconst \1 = compiler.\2;',
        content
    )
    
    # 替换运行时导入
    content = re.sub(
        r'const (\w+) = @import\("\.\.\/runtime\/(\w+)\.zig"\);',
        r'const runtime = @import("runtime");\nconst \1 = runtime.\2;',
        content
    )
    
    # 替换字节码导入
    content = re.sub(
        r'const (\w+) = @import\("\.\.\/bytecode\/(\w+)\.zig"\);',
        r'const bytecode = @import("bytecode");\nconst \1 = bytecode.\2;',
        content
    )
    
    # 替换 JIT 导入
    content = re.sub(
        r'const (\w+) = @import\("\.\.\/jit\/(\w+)\.zig"\);',
        r'const jit = @import("jit");\nconst \1 = jit.\2;',
        content
    )
    
    # 替换扩展导入
    content = re.sub(
        r'const (\w+) = @import\("\.\.\/extension\/(\w+)\.zig"\);',
        r'const extension = @import("extension");\nconst \1 = extension.\2;',
        content
    )
    
    with open(file_path, 'w') as f:
        f.write(content)

# 处理所有需要修复的文件
files_to_fix = [
    "src/bytecode/vm.zig",
    "src/bytecode/generator.zig",
    # ... 其他文件
]

for file_path in files_to_fix:
    if os.path.exists(file_path):
        fix_imports(file_path)
        print(f"✅ 修复: {file_path}")
```
