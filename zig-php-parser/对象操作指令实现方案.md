# AOT 编译器对象操作指令实现方案

## 任务概述
为 AOT 编译器添加对象操作指令支持，使其能够编译 PHP 的面向对象特性。

## IR 指令定义（已存在）
在 `src/aot/ir.zig` 中已定义：
- `new_object: NewObjectOp` - 创建对象
- `property_get: PropertyGetOp` - 获取属性
- `property_set: PropertySetOp` - 设置属性
- `method_call: MethodCallOp` - 调用方法

## 实现步骤

### 1. 在 native_linker.zig 的 generateInstructionSimple 方法中添加指令处理

#### 1.1 new_object 指令
```zig
.new_object => |op| {
    if (inst.result) |reg| {
        // 生成参数列表
        var args_buf = std.ArrayList(u8){};
        defer args_buf.deinit(self.allocator);
        const args_writer = args_buf.writer(self.allocator);
        
        for (op.args, 0..) |arg, i| {
            if (i > 0) try args_writer.writeAll(", ");
            const arg_type_tag = @as(std.meta.Tag(IR.Type), arg.type_);
            if (arg_type_tag == .i64) {
                try args_writer.print("runtime.Value.initInt(reg_{d})", .{arg.id});
            } else if (arg_type_tag == .f64) {
                try args_writer.print("runtime.Value.initFloat(reg_{d})", .{arg.id});
            } else if (arg_type_tag == .bool) {
                try args_writer.print("runtime.Value.initBool(reg_{d})", .{arg.id});
            } else {
                try args_writer.print("reg_{d}", .{arg.id});
            }
        }
        
        try writer.print("    reg_{d} = try runtime.Value.initObject(\"{s}\", runtime.runtime_allocator);\n", 
            .{reg.id, op.class_name});
    }
},
```

#### 1.3 property_set 指令
```zig
.property_set => |op| {
    const value_type_tag = @as(std.meta.Tag(IR.Type), op.value.type_);
    
    const value_expr = if (value_type_tag == .php_value)
        try std.fmt.allocPrint(self.allocator, "reg_{d}", .{op.value.id})
    else if (value_type_tag == .i64)
        try std.fmt.allocPrint(self.allocator, "runtime.Value.initInt(reg_{d})", .{op.value.id})
    else if (value_type_tag == .f64)
        try std.fmt.allocPrint(self.allocator, "runtime.Value.initFloat(reg_{d})", .{op.value.id})
    else if (value_type_tag == .bool)
        try std.fmt.allocPrint(self.allocator, "runtime.Value.initBool(reg_{d})", .{op.value.id})
    else
        try std.fmt.allocPrint(self.allocator, "reg_{d}", .{op.value.id});
    defer self.allocator.free(value_expr);
    
    try writer.print("    try runtime.php_object_set(reg_{d}, \"{s}\", {s});\n", 
        .{op.object.id, op.property_name, value_expr});
},
```

### 2. 在 runtime_lib_template.zig 中添加运行时支持

#### 2.1 对象结构定义
```zig
/// PHP对象类型
pub const PHPObject = struct {
    class_name: []const u8,
    properties: std.StringHashMap(Value),
    ref_count: usize,
    
    pub fn init(allocator: Allocator, class_name: []const u8) !*PHPObject {
        const obj = try allocator.create(PHPObject);
        obj.class_name = try allocator.dupe(u8, class_name);
        obj.properties = std.StringHashMap(Value).init(allocator);
        obj.ref_count = 1;
        return obj;
    }
    
    pub fn retain(self: *PHPObject) void {
        self.ref_count += 1;
    }
    
    pub fn release(self: *PHPObject, allocator: Allocator) void {
        if (self.ref_count == 0) return;
        self.ref_count -= 1;
        if (self.ref_count == 0) {
            self.deinit(allocator);
        }
    }
    
    fn deinit(self: *PHPObject, allocator: Allocator) void {
        var iter = self.properties.iterator();
        while (iter.next()) |entry| {
            entry.value_ptr.release(allocator);
        }
        self.properties.deinit();
        allocator.free(self.class_name);
        allocator.destroy(self);
    }
};
```

#### 2.2 Value 类型扩展
在 Value 结构中添加对象类型支持：
```zig
pub const TYPE_OBJECT: u64 = 0x0001800000000000;

pub fn initObject(obj: *PHPObject) Value {
    const addr = @intFromPtr(obj);
    return .{ .val = TAG_PTR | TYPE_OBJECT | (addr & 0x00007FFFFFFFFFFF) };
}

pub fn isObject(self: Value) bool {
    if ((self.val & (SIGN_BIT | QNAN)) != QNAN) return false;
    return (self.val & TYPE_MASK) == TYPE_OBJECT;
}

pub fn asObject(self: Value) *PHPObject {
    return @ptrFromInt(self.val & 0x00007FFFFFFFFFFF);
}
```

#### 2.3 运行时函数
```zig
/// 创建对象
pub fn php_object_new(class_name: []const u8, allocator: Allocator) !Value {
    const obj = try PHPObject.init(allocator, class_name);
    return Value.initObject(obj);
}

/// 获取对象属性
pub fn php_object_get(obj_val: Value, property_name: []const u8) !Value {
    if (!obj_val.isObject()) return Value.initNull();
    const obj = obj_val.asObject();
    return obj.properties.get(property_name) orelse Value.initNull();
}

/// 设置对象属性
pub fn php_object_set(obj_val: Value, property_name: []const u8, value: Value) !void {
    if (!obj_val.isObject()) return error.NotAnObject;
    const obj = obj_val.asObject();
    
    // 释放旧值
    if (obj.properties.get(property_name)) |old_value| {
        old_value.release(runtime_allocator);
    }
    
    // 保留新值
    _ = value.retain();
    
    try obj.properties.put(property_name, value);
}

/// 调用对象方法（简化版）
pub fn php_object_call(obj_val: Value, method_name: []const u8, args: []const Value) !Value {
    _ = obj_val;
    _ = method_name;
    _ = args;
    // 简化实现：暂时返回null
    // 完整实现需要方法查找和调用机制
    return Value.initNull();
}
```

## 注意事项

1. **内存管理**：对象使用引用计数，需要正确处理 retain/release
2. **类型检查**：在运行时函数中添加类型检查，防止非对象值的操作
3. **方法调用**：完整的方法调用需要类定义和方法查找机制，当前实现为简化版
4. **构造函数**：需要特殊处理 `__construct` 方法
5. **继承和多态**：当前不支持，需要后续扩展

## 实现优先级

1. ✅ 对象创建 (new_object)
2. ✅ 属性读取 (property_get)
3. ✅ 属性设置 (property_set)
4. ⚠️ 方法调用 (method_call) - 简化实现
