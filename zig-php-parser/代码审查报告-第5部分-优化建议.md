// src/bytecode/vm.zig

const CompleteGC = @import("../runtime/complete_gc.zig").CompleteGC;

pub const BytecodeVM = struct {
    allocator: std.mem.Allocator,
    
    // 移除简化的 GC 字段
    // string_pool: std.ArrayListUnmanaged(*Value.String),
    // array_pool: std.ArrayListUnmanaged(*Value.Array),
    // object_pool: std.ArrayListUnmanaged(*Value.Object),
    
    // 使用完整的 GC
    gc: CompleteGC,
    
    // ... 其他字段
    
    pub fn init(allocator: std.mem.Allocator) !*BytecodeVM {
        const vm = try allocator.create(BytecodeVM);
        vm.* = BytecodeVM{
            .allocator = allocator,
            .gc = try CompleteGC.init(allocator),
            // ... 其他初始化
        };
        return vm;
    }
    
    pub fn deinit(self: *BytecodeVM) void {
        self.gc.deinit();
        // ... 其他清理
    }
    
    /// 创建字符串 - 使用 GC 分配
    pub fn createString(self: *BytecodeVM, data: []const u8) !*Value.String {
        const obj = try self.gc.allocate(
            @sizeOf(Value.String) + data.len,
            .string
        );
        
        const str = @as(*Value.String, @ptrCast(@alignCast(obj.data.ptr)));
        str.* = .{
            .data = obj.data[@sizeOf(Value.String)..],
            .ref_count = 1,
            .marked = false,
        };
        @memcpy(str.data, data);
        
        // 添加为 GC 根（如果需要）
        const value = Value{ .string_val = str };
        try self.gc.addRoot(&value);
        
        return str;
    }
    
    /// 创建数组 - 使用 GC 分配
    pub fn createArray(self: *BytecodeVM) !*Value.Array {
        const obj = try self.gc.allocate(
            @sizeOf(Value.Array),
            .array
        );
        
        const arr = @as(*Value.Array, @ptrCast(@alignCast(obj.data.ptr)));
        arr.* = .{
            .elements = .{},
            .keys = .{},
            .ref_count = 1,
            .marked = false,
        };
        
        return arr;
    }
    
    /// 触发 GC
    pub fn collectGarbage(self: *BytecodeVM) !void {
        try self.gc.collect();
    }
};

### 3. 实现 GC 压缩阶段

**问题**：vm.zig 的 compactPhase 只是打桩，不实际执行

**影响面**：内存效率，长时间运行后碎片化

**落地成本**：低（已有 complete_gc.zig 实现）

**修复方案**：

```zig
// complete_gc.zig 已有实现，需要完善 updateReferences

fn updateReferences(
    self: *CompleteGC,
    obj: *GCObject,
    new_addresses: *std.AutoHashMap(*GCObject, [*]u8)
) !void {
    switch (obj.obj_type) {
        .array => {
            const arr = @as(*types.PHPArray, @ptrCast(@alignCast(obj.data.ptr)));
            var iter = arr.map.iterator();
            while (iter.next()) |entry| {
                try self.updateValueReference(entry.value_ptr, new_addresses);
            }
        },
        
        .object => {
            const php_obj = @as(*types.PHPObject, @ptrCast(@alignCast(obj.data.ptr)));
            var iter = php_obj.properties.iterator();
            while (iter.next()) |entry| {
                try self.updateValueReference(entry.value_ptr, new_addresses);
            }
        },
        
        .closure => {
            const closure = @as(*types.PHPClosure, @ptrCast(@alignCast(obj.data.ptr)));
            for (closure.captured_vars.items) |*captured_value| {
                try self.updateValueReference(captured_value, new_addresses);
            }
        },
        
        else => {},
    }
}

fn updateValueReference(
    self: *CompleteGC,
    value: *Value,
    new_addresses: *std.AutoHashMap(*GCObject, [*]u8)
) !void {
    switch (value.*) {
        .string_val => |*str_ptr| {
            const old_obj = @fieldParentPtr(GCObject, "data", str_ptr.*.data.ptr);
            if (new_addresses.get(old_obj)) |new_addr| {
                const new_str = @as(*Value.String, @ptrCast(@alignCast(new_addr)));
                str_ptr.* = new_str;
            }
        },
        
        .array_val => |*arr_ptr| {
            const old_obj = @fieldParentPtr(GCObject, "data", @as([*]u8, @ptrCast(arr_ptr.*)));
            if (new_addresses.get(old_obj)) |new_addr| {
                const new_arr = @as(*Value.Array, @ptrCast(@alignCast(new_addr)));
                arr_ptr.* = new_arr;
            }
        },
        
        // 其他类型类似处理
        else => {},
    }
}
```

### 4. 完善函数调用的参数处理

**问题**：vm.zig 的 call 方法中参数处理不完整

**影响面**：函数调用功能

**落地成本**：中等（1-2天）

**修复方案**：

```zig
// src/bytecode/vm.zig

/// 参数传递模式
pub const ParamPassMode = enum {
    by_value,      // 值传递
    by_reference,  // 引用传递
};

/// 参数信息
pub const ParamInfo = struct {
    name: []const u8,
    pass_mode: ParamPassMode,
    default_value: ?Value,
    is_variadic: bool,
};

/// 完整的函数调用实现
pub fn call(self: *BytecodeVM, name: []const u8, args: []const Value) VMError!Value {
    const func = self.functions.get(name) orelse return VMError.UndefinedFunction;
    
    // 1. 验证参数数量
    const min_args = func.min_args;
    const max_args = if (func.is_variadic) std.math.maxInt(usize) else func.max_args;
    
    if (args.len < min_args or args.len > max_args) {
        return VMError.InvalidArgumentCount;
    }
    
    // 2. 创建调用帧
    if (self.frame_count >= FRAMES_MAX) {
        return VMError.StackOverflow;
    }
    
    const frame_idx = self.frame_count;
    self.frame_count += 1;
    
    self.frames[frame_idx] = CallFrame{
        .function = func,
        .ip = 0,
        .base_pointer = self.stack_top,
        .return_address = 0,
    };
    
    // 3. 处理参数
    var param_idx: usize = 0;
    while (param_idx < func.param_count) : (param_idx += 1) {
        const param_info = func.param_info[param_idx];
        
        if (param_idx < args.len) {
            // 有实参
            const arg = args[param_idx];
            
            switch (param_info.pass_mode) {
                .by_value => {
                    // 值传递：深拷贝
                    const copied = try self.deepCopyValue(arg);
                    try self.push(copied);
                },
                
                .by_reference => {
                    // 引用传递：创建引用包装
                    const ref = try self.createReference(arg);
                    try self.push(ref);
                },
            }
        } else {
            // 无实参，使用默认值
            if (param_info.default_value) |default| {
                try self.push(default);
            } else {
                return VMError.InvalidArgumentCount;
            }
        }
    }
    
    // 4. 处理可变参数
    if (func.is_variadic and args.len > func.param_count) {
        const variadic_array = try self.createArray();
        
        var i = func.param_count;
        while (i < args.len) : (i += 1) {
            try self.arrayPush(variadic_array, args[i]);
        }
        
        try self.push(Value{ .array_val = variadic_array });
    }
    
    // 5. 分配局部变量
    var i: u32 = 0;
    while (i < func.local_count) : (i += 1) {
        try self.push(.null_val);
    }
    
    // 6. 执行函数
    const result = try self.executeFrame(&self.frames[frame_idx]);
    
    // 7. 清理
    self.frame_count -= 1;
    
    return result;
}

/// 深拷贝值
fn deepCopyValue(self: *BytecodeVM, value: Value) !Value {
    return switch (value) {
        .null_val, .bool_val, .int_val, .float_val => value,
        
        .string_val => |str| blk: {
            const new_str = try self.createString(str.data);
            break :blk Value{ .string_val = new_str };
        },
        
        .array_val => |arr| blk: {
            const new_arr = try self.createArray();
            for (arr.elements.items) |elem| {
                const copied_elem = try self.deepCopyValue(elem);
                try self.arrayPush(new_arr, copied_elem);
            }
            break :blk Value{ .array_val = new_arr };
        },
        
        // 其他类型类似处理
        else => value,
    };
}

/// 创建引用包装
fn createReference(self: *BytecodeVM, value: Value) !Value {
    // 实现引用语义
    // 可以使用指针或引用计数
    _ = self;
    return value; // 简化：直接返回值
}
```

// src/bytecode/vm.zig

/// 结构体定义
pub const StructDef = struct {
    name: []const u8,
    fields: []FieldDef,
    methods: std.StringHashMap(*CompiledFunction),
    
    pub const FieldDef = struct {
        name: []const u8,
        type_hint: ?TypeInfo,
        default_value: ?Value,
    };
};

pub const BytecodeVM = struct {
    // ... 其他字段
    
    /// 结构体定义表
    struct_defs: std.StringHashMap(StructDef),
    
    pub fn init(allocator: std.mem.Allocator) !*BytecodeVM {
        // ...
        .struct_defs = std.StringHashMap(StructDef).init(allocator),
        // ...
    }
    
    /// 注册结构体定义
    pub fn registerStruct(
        self: *BytecodeVM,
        name: []const u8,
        fields: []const StructDef.FieldDef,
        methods: std.StringHashMap(*CompiledFunction)
    ) !void {
        const struct_def = StructDef{
            .name = try self.allocator.dupe(u8, name),
            .fields = try self.allocator.dupe(StructDef.FieldDef, fields),
            .methods = methods,
        };
        
        try self.struct_defs.put(name, struct_def);
    }
    
    /// 查找结构体方法
    fn lookupStructMethod(
        self: *BytecodeVM,
        struct_instance: *Value.StructInstance,
        method_name: []const u8
    ) !*CompiledFunction {
        // 1. 获取结构体定义
        const struct_id = struct_instance.struct_id;
        
        // 2. 从结构体定义表查找
        var iter = self.struct_defs.iterator();
        while (iter.next()) |entry| {
            const struct_def = entry.value_ptr;
            
            // 简化：使用名称匹配（实际应使用 ID）
            if (struct_def.methods.get(method_name)) |method| {
                return method;
            }
        }
        
        return VMError.UndefinedFunction;
    }
    
    /// 调用结构体方法
    pub fn callStructMethod(
        self: *BytecodeVM,
        struct_instance: *Value.StructInstance,
        method_name: []const u8,
        args: []const Value
    ) !Value {
        const method = try self.lookupStructMethod(struct_instance, method_name);
        
        // 将 self 作为第一个参数
        var full_args = try self.allocator.alloc(Value, args.len + 1);
        defer self.allocator.free(full_args);
        
        full_args[0] = Value{ .struct_val = struct_instance };
        @memcpy(full_args[1..], args);
        
        return self.callFunction(method, full_args);
    }
};
