**实现位置**:

**核心功能**:
```zig
// 正则编译与缓存
fn getOrCompileRegex(pattern: []const u8, options: c_uint, allocator: Allocator) !*pcre2_code {
    // LRU缓存查找
    regex_cache_mutex.lock();
    defer regex_cache_mutex.unlock();
    
    if (regex_cache.getPtr(pattern)) |entry| {
        entry.last_used = std.time.nanoTimestamp();
        return entry.code;
    }
    
    // 编译新模式
    const re = pcre2_compile_8(pattern.ptr, pattern.len, options, &errcode, &erroffset, null);
    
    // LRU淘汰
    if (regex_cache.count() >= REGEX_CACHE_SIZE) {
        evictOldest();
    }
    
    try regex_cache.put(pattern_copy, .{ .code = re, .last_used = now });
    return re;
}
```

### 3.2 引用参数机制

**问题**: PHP的`preg_match($pattern, $subject, &$matches)`需要写回matches数组

**解决方案**:

#### IR生成阶段 (ir_generator.zig)
```zig
// 检测引用参数
if (matches_node.tag == .variable) {
    const var_name = getString(matches_node.data.variable.name);
    const is_global = global_vars.contains(var_name);
    
    if (is_global) {
        // 全局变量：创建临时alloca
        const temp_reg = alloca(php_value);
        store(global_get(var_name), temp_reg);
        const ref_reg = make_ref(temp_reg);
        
        // 记录需要写回
        ref_writebacks.append(.{ .var_name = var_name, .temp_reg = temp_reg });
    } else {
        // 局部变量：直接使用alloca
        const var_reg = getOrCreateVarRegister(var_name, .php_value);
        const ref_reg = make_ref(var_reg);
    }
}

// 函数调用后写回
for (ref_writebacks.items) |wb| {
    const new_val = load(wb.temp_reg);
    global_set(wb.var_name, new_val);
}
```

#### 代码生成阶段 (native_linker.zig)
```zig
// 查找make_ref的源alloca
fn findMakeRefSource(reg: Register) ?Register {
    const inst = getInstruction(reg);
    if (inst.op == .make_ref) {

// 生成函数调用
if (runtime_name == "preg_match_with_matches") {
```

### 3.3 性能优化

#### LRU缓存策略
```zig
const RegexCacheEntry = struct {
    code: *pcre2_code,
    last_used: i128,
};

var regex_cache: std.StringHashMap(RegexCacheEntry) = undefined;
var regex_cache_mutex: std.Thread.Mutex = .{};

const REGEX_CACHE_SIZE = 128;

fn evictOldest() void {
    var oldest_key: ?[]const u8 = null;
    var oldest_time: i128 = std.math.maxInt(i128);
    
    var iter = regex_cache.iterator();
    while (iter.next()) |entry| {
        if (entry.value_ptr.last_used < oldest_time) {
            oldest_time = entry.value_ptr.last_used;
            oldest_key = entry.key_ptr.*;
        }
    }
    
    if (oldest_key) |key| {
        const entry = regex_cache.get(key).?;
        pcre2_code_free_8(entry.code);
        _ = regex_cache.remove(key);
    }
}
```

#### 零拷贝优化

**问题**: 引用参数传递机制不完整  
**影响**: preg_match/preg_match_all的matches参数无法正确填充  
**范围**: 仅Tree模式（调试模式）

**解决方案**: 改进Tree模式的引用参数处理（预计2天工作量）

### 5.2 编译时验证未实现

**原因**: 模块依赖冲突（ir_generator不能导入runtime模块）  
**影响**: 无法在编译时检测无效正则  
**替代方案**: 运行时验证（PCRE2编译失败时返回错误）  
**优先级**: 低（大多数正则是动态的）  
**解决方案**: 重构模块依赖或使用独立验证模块（预计1周工作量）

### 5.3 preg_grep限制

**限制**: AOT实现仅支持数字索引数组  
**影响**: 关联数组的键不会保留  
**范围**: 仅AOT模式  
**优先级**: 低（大多数用例使用数字索引）  
**解决方案**: 实现完整的关联数组支持（预计1天工作量）

## 八、API文档

### 8.1 preg_match

**签名**: `preg_match(string $pattern, string $subject, array &$matches = null, int $flags = 0, int $offset = 0): int|false`

**功能**: 执行正则表达式匹配

**参数**:
- `$pattern`: 正则表达式模式
- `$subject`: 输入字符串
- `$matches`: 可选，匹配结果数组（引用参数）
- `$flags`: 可选，匹配标志
- `$offset`: 可选，开始位置

**返回值**: 匹配次数（0或1），失败返回false

**示例**:
```php
$text = "Hello World 123";
$matches = [];
$ret = preg_match('/(\w+)\s+(\w+)\s+(\d+)/', $text, $matches);
// $ret = 1
// $matches = ["Hello World 123", "Hello", "World", "123"]
```

### 8.2 preg_match_all

**签名**: `preg_match_all(string $pattern, string $subject, array &$matches = null, int $flags = PREG_PATTERN_ORDER, int $offset = 0): int|false`

**功能**: 执行全局正则表达式匹配

**参数**:
- `$pattern`: 正则表达式模式
- `$subject`: 输入字符串
- `$matches`: 可选，匹配结果数组（引用参数）
- `$flags`: 可选，结果格式（PREG_PATTERN_ORDER或PREG_SET_ORDER）
- `$offset`: 可选，开始位置

**返回值**: 匹配次数，失败返回false

**示例**:
```php
$text = "test123abc456";
$matches = [];
$count = preg_match_all('/([a-z]+)(\d+)/', $text, $matches);
// $count = 2
// $matches[0] = ["test123", "abc456"]
// $matches[1] = ["test", "abc"]
// $matches[2] = ["123", "456"]
```

### 8.3 preg_replace

**签名**: `preg_replace(string|array $pattern, string|array $replacement, string|array $subject, int $limit = -1, int &$count = null): string|array|null`

**功能**: 执行正则表达式替换

**参数**:
- `$pattern`: 正则表达式模式
- `$replacement`: 替换字符串
- `$subject`: 输入字符串或数组
- `$limit`: 可选，最大替换次数
- `$count`: 可选，实际替换次数（引用参数）

**返回值**: 替换后的字符串或数组

**示例**:
```php
$text = "test123abc456";
$result = preg_replace('/\d+/', 'X', $text);
// $result = "testXabcX"
```

### 8.4 preg_split

**签名**: `preg_split(string $pattern, string $subject, int $limit = -1, int $flags = 0): array|false`

**功能**: 通过正则表达式分割字符串

**参数**:
- `$pattern`: 正则表达式模式
- `$subject`: 输入字符串
- `$limit`: 可选，最大分割次数
- `$flags`: 可选，分割标志

**返回值**: 分割后的数组

**示例**:
```php
$text = "test123abc456";
$parts = preg_split('/\d+/', $text);
// $parts = ["test", "abc", ""]
```

### 8.5 preg_grep

**签名**: `preg_grep(string $pattern, array $array, int $flags = 0): array|false`

**功能**: 返回匹配正则表达式的数组元素

**参数**:
- `$pattern`: 正则表达式模式
- `$array`: 输入数组
- `$flags`: 可选，PREG_GREP_INVERT = 1（反向匹配）

**返回值**: 过滤后的数组

**示例**:
```php
$array = ['apple', 'banana', 'cherry'];
$result = preg_grep('/a/', $array);
// $result = ['apple', 'banana']

$result2 = preg_grep('/a/', $array, 1);  // 反向
// $result2 = ['cherry']
```
