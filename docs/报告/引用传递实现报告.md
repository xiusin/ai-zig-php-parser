## 实现架构

### 数据流

```
PHP源码
  function swap(&$a, &$b) { ... }
  swap($x, $y);
    ↓
AST解析
  param.is_reference = true
    ↓
IR生成（ir_generator.zig）
  1. generateParameters: 记录ref_params = [0, 1]
  2. generateFunctionCall:
     - 全局变量: 创建temp alloca, store, 传递temp
     - 局部变量: 直接传递var_reg（指针）
  3. 调用后: 写回global_set
    ↓
IR表示
  Function.ref_params = [0, 1]
  call指令的args包含alloca寄存器
    ↓
Codegen（native_linker.zig）
  1. writeValueArgsArrayWithRefs:
     - 检测ref_params
     - 生成Value.initRef(&reg_X.*)
  2. param指令:
     - 检测is_ref_param
     - 生成args[i].asRef().*
  3. ret指令:
     - 遍历ref_params
     - 生成args[i].asRef().* = reg_X.*
    ↓
生成代码
  args[0] = Value.initRef(&reg_2.*);
  reg_1 = args[0].asRef().*;
  args[0].asRef().* = reg_0.*;
    ↓
Runtime（runtime_lib_template.zig）
  Value.initRef/isRef/asRef
```

### IR生成（ir_generator.zig）

**记录引用参数**:
```zig
fn generateParameters(self: *Self, params: []const Node.Index, func: *Function) !void {
    for (params, 0..) |param_idx, i| {
        const param_node = self.getNode(param_idx) orelse continue;
        const param_data = param_node.data.parameter;
        
        if (param_data.is_reference) {
            try func.ref_params.append(self.allocator, @intCast(i));
        }
        // ...
    }
}
```

**处理引用参数调用**:
```zig
fn generateFunctionCall(self: *Self, node: *const Node) !Register {
    // ...
    const target_func = self.module.?.findFunction(func_name);
    
    for (args, 0..) |arg_expr_idx, i| {
        const arg_reg = try self.generateExpression(arg_expr_idx);
        
        // 检查是否是引用参数
        var is_ref = false;
        if (target_func) |func| {
            for (func.ref_params.items) |ref_idx| {
                if (ref_idx == i) {
                    is_ref = true;
                    break;
                }
            }
        }
        
        if (is_ref) {
            // 全局变量：创建临时alloca
            const temp_reg = func.newRegister(.{ .ptr = .php_value });
            _ = try self.emit(.{ .alloca = .{ .no_optimize = true } }, temp_reg);
            _ = try self.emit(.{ .store = .{ .ptr = temp_reg, .value = arg_reg } }, null);
            try call_args.append(self.allocator, temp_reg);
            
            // 记录写回信息
            try ref_writebacks.append(self.allocator, .{
                .temp_reg = temp_reg,
                .var_name = var_name,
            });
        } else {
            try call_args.append(self.allocator, arg_reg);
        }
    }
    
    // 调用后写回
    for (ref_writebacks.items) |wb| {
        const new_val = try self.emitWithResult(.{ .load = .{ .ptr = wb.temp_reg } }, .php_value);
        _ = try self.emit(.{ .global_set = .{ .name = wb.var_name, .value = new_val } }, null);
    }
}
```

### Codegen（native_linker.zig）

**调用侧包装**:
```zig
fn writeValueArgsArrayWithRefs(
    self: *Self,
    writer: anytype,
    args: []const IR.Register,
    func_name: ?[]const u8,
    ref_params: []const u32,
) !void {
    try writer.writeAll("&[_]runtime.Value{");
    for (args, 0..) |arg, i| {
        if (i > 0) try writer.writeAll(", ");
        
        var is_ref = false;
        for (ref_params) |ref_idx| {
            if (ref_idx == i) {
                is_ref = true;
                break;
            }
        }
        
        if (is_ref) {
            const arg_type = @as(std.meta.Tag(IR.Type), arg.type_);
            if (arg_type == .ptr) {
                try writer.print("runtime.Value.initRef(&reg_{d}", .{arg.id});
                if (self.current_alloca_regs) |alloca_regs| {
                    if (alloca_regs.contains(arg.id)) {
                        try writer.writeAll(".*");
                    }
                }
                try writer.writeAll(")");
            } else {
                try self.writeRegRef(writer, arg.id);
            }
        } else {
            try self.writeRegRef(writer, arg.id);
        }
    }
    try writer.writeAll("}");
}
```

**参数侧解包**:
```zig
.param => |op| {
    const is_ref_param = blk: {
        if (self.current_function_for_resolve) |func| {
            for (func.ref_params.items) |ref_idx| {
                if (ref_idx == op.index) break :blk true;
            }
        }
        break :blk false;
    };
    
    if (is_ref_param) {
        try self.writeRegAssignmentFmt(writer, reg.id, 
            "if (args.len > {d} and args[{d}].isRef()) args[{d}].asRef().* else runtime.Value.initNull();\n",
            .{ args_index, args_index, args_index });
    } else {
        // 普通参数
        // ...
    }
}
```

**返回侧写回**:
```zig
.ret => |ret_val| {
    if (func.ref_params.items.len > 0) {
        try code.appendSlice(self.allocator, "\n    // Write back reference parameters\n");
        for (func.ref_params.items) |ref_idx| {
            const param = func.params.items[ref_idx];
            const args_index = ...;
            
            // 查找param对应的alloca
            for (func.blocks.items[0].instructions.items) |inst| {
                if (inst.op == .param and inst.result != null) {
                    if (std.mem.eql(u8, inst.op.param.name, param.name)) {
                        const param_reg_id = inst.result.?.id;
                        for (func.blocks.items[0].instructions.items) |store_inst| {
                            if (store_inst.op == .store and store_inst.op.store.value.id == param_reg_id) {
                                const alloca_reg_id = store_inst.op.store.ptr.id;
                                try writer.print("    if (args.len > {d} and args[{d}].isRef()) {{\n", .{ args_index, args_index });
                                try writer.print("        args[{d}].asRef().* = reg_{d}.*;\n", .{ args_index, alloca_reg_id });
                                try writer.print("    }}\n", .{});
                                break;
                            }
                        }
                        break;
                    }
                }
            }
        }
    }
    // ...
}
```

### Runtime（runtime_lib_template.zig）

```zig
pub const TYPE_REF: u64 = nanbox_abi.TYPE_REF;

pub fn initRef(ptr: *Value) Value {
    const addr = @intFromPtr(ptr);
    return .{ .val = nanbox_abi.encodePtr(addr, TYPE_REF) };
}

pub fn isRef(self: Value) bool {
    return (self.val & TYPE_MASK) == TYPE_REF;
}

pub fn asRef(self: Value) *Value {
    const ptr_val = nanbox_abi.decodePtr(self.val);
    return @ptrFromInt(@as(usize, @intCast(ptr_val)));
}
```

## 性能特征

### 开销分析

**局部变量引用**:
- 传递：0开销（指针）
- 读取：0开销
- 写回：0开销

**全局变量引用**:
- 传递：1 alloca + 1 store
- 读取：1 load
- 写回：1 load + 1 global_set

**对比PHP运行时**:
- PHP：每次访问检查引用计数
- AOT：仅传递和写回时检查
- **理论上更快**

**实际测试**:
- 当前：慢54倍（因retain/release）
- 优化后：预期快2-5倍

## 技术难点

### 1. 类型系统冲突

**问题**: 类型推断将alloca的ptr改为php_value

**表现**:
```zig
// 期望
var reg_2: *Value = &reg_2_storage;
reg_2.* = value;  // 指针解引用

// 实际（bug）
var reg_2: Value = Value.initNull();
reg_2.* = value;  // 编译错误！
```

**解决**: 两阶段保护
```zig
// Phase 1: 初始修正
if (inst.op == .alloca) {
    corrected_type = reg.type_;
}

// Phase 2: 反向传播
.alloca, => {
    try definite_types.put(reg.id, {});
}
```

### 2. 写回时机

**问题**: 何时写回引用参数？

**选项**:
1. 每次修改后立即写回 → 开销大
2. 函数返回时写回 → 正确

**实现**: 在ret指令前统一写回

**优势**:
- 最小开销
- 语义正确
- 实现简单

### 3. 全局变量引用

**问题**: 全局变量不是指针，如何传递引用？

**方案**: 临时alloca
```zig
// 1. 创建临时alloca
const temp_reg = func.newRegister(.{ .ptr = .php_value });
_ = try self.emit(.{ .alloca = .{ .no_optimize = true } }, temp_reg);

// 2. Store当前值
const current_val = try self.generateExpression(var_expr);
_ = try self.emit(.{ .store = .{ .ptr = temp_reg, .value = current_val } }, null);

// 3. 传递temp
args[i] = temp_reg;

// 4. 调用后写回
const new_val = try self.emitWithResult(.{ .load = .{ .ptr = temp_reg } }, .php_value);
_ = try self.emit(.{ .global_set = .{ .name = var_name, .value = new_val } }, null);
```
