
```zig
// 1. 生成唯一类名并intern
const name_temp = try std.fmt.allocPrint(..., "anonymous_class_{d}", .{id});
const name_sid = try module.internString(name_temp);
const name = module.getString(name_sid);

// 2. 创建TypeDef
const type_def = try allocator.create(TypeDef);
type_def.* = .{
    .name = name,
    .kind = .class,
    .properties = &.{},
    .methods = &.{},
    ...
};

// 3. 注册类
try module.addTypeDef(type_def);
try symbol_table.defineClass(name, ...);

// 4. 处理成员
_ = try symbol_table.enterScope(.class, name);
for (members) |member| {
    switch (member.tag) {
        .method_decl => try generateMethodDecl(member, name),
        .property_decl => try generatePropertyDecl(member, name),
        else => {},
    }
}
symbol_table.leaveScope();

// 5. 实例化
return emitWithResult(.{ .new_object = .{
    .class_name = name,
    .args = args_slice,
} }, .{ .php_object = name });
```

**关键修复**:
1. 字符串生命周期：通过internString管理
2. args生命周期：通过allocator.dupe管理
3. 属性注册：添加到TypeDef.properties
4. builtin_map：添加php_object_new系列函数

### 3. MT19937随机数生成器 ✅

**算法实现**:
```zig
const MT19937 = struct {
    state: [624]u32,
    index: usize,
    
    fn init(seed: u32) MT19937 {
        // 标准MT19937初始化
        mt.state[0] = seed;
        for (1..624) |i| {
            mt.state[i] = 1812433253 *% (mt.state[i-1] ^ (mt.state[i-1] >> 30)) +% i;
        }
        return mt;
    }
    
    fn generate(self: *MT19937) u32 {
        if (self.index >= 624) self.twist();
        var y = self.state[self.index];
        self.index += 1;
        
        // Tempering
        y ^= (y >> 11);
        y ^= (y << 7) & 0x9D2C5680;
        y ^= (y << 15) & 0xEFC60000;
        y ^= (y >> 18);
        
        return y;
    }
    
    fn twist(self: *MT19937) void {
        // 标准twist算法
        for (0..624) |i| {
            const x = (self.state[i] & 0x80000000) + (self.state[(i+1)%624] & 0x7FFFFFFF);
            var xA = x >> 1;
            if ((x & 1) != 0) xA ^= 0x9908B0DF;
            self.state[i] = self.state[(i+397)%624] ^ xA;
        }
        self.index = 0;
    }
};

threadlocal var mt_state: ?MT19937 = null;
```

**特性**:
- PHP兼容输出
- 线程安全（threadlocal）
- 自动初始化
- 标准算法实现

**测试**: test_036_math_funcs ✅

## 技术架构

### 引用传递数据流

```
PHP代码: function swap(&$a, &$b)
  ↓
IR生成: Function.ref_params = [0, 1]
  ↓
IR指令: 
  - 调用侧: 创建temp alloca, store, 传递temp
  - 函数侧: param指令标记引用
  - 返回侧: 写回到args[i].asRef().*
  ↓
Codegen:
  - 调用侧: Value.initRef(&reg_X.*)
  - 函数侧: args[i].asRef().*
  - 返回侧: args[i].asRef().* = reg_X.*
  ↓
Runtime: Value.initRef/isRef/asRef
```

### 匿名类生命周期

```
PHP代码: new class { ... }
  ↓
IR生成:
  1. 生成唯一类名
  2. internString确保持久化
  3. 创建TypeDef
  4. 注册到module.types
  5. 处理成员（方法、属性）
  6. 生成new_object指令
  ↓
Codegen:
  1. 生成ClassMeta.init
  2. 添加方法（addMethod）
  3. 添加属性（addProperty）
  4. 注册类（registerClass）
  5. 调用php_object_new_with_constructor
  ↓
Runtime: 标准对象创建流程
```

## 关键技术决策

### 1. 引用传递：运行时包装 vs 编译时指针

**选择**: 运行时包装（Value.initRef）

**理由**:
- PHP是动态类型，引用可以指向任何类型
- 编译时无法确定所有引用关系
- 运行时包装提供最大灵活性

**代价**: 
- 每次引用传递需要initRef/asRef调用
- 约4条额外指令

**优势**:
- 类型安全
- 支持所有PHP引用场景
- 实现简单清晰

### 2. 匿名类：运行时生成 vs 编译时展开

**选择**: 编译时展开

**理由**:
- 匿名类在编译时已知
- 可以生成优化的静态代码
- 避免运行时类型创建开销

**实现**:
- 编译时生成唯一类名
- 编译时注册TypeDef
- 运行时直接使用预注册类

**优势**:
- 零运行时开销
- 完整类型信息
- 支持所有OOP特性

### 3. MT19937：自实现 vs 调用libc

**选择**: 自实现

**理由**:
- PHP使用MT19937，libc使用其他算法
- 需要与PHP输出完全一致
- MT19937算法简单（~50行）

**实现**:
- 624个32位状态
- 标准twist算法
- 完整tempering步骤

**优势**:
- PHP兼容
- 可预测输出
- 无外部依赖

**问题**: 类型推断会破坏alloca的ptr类型

- 全局变量：临时alloca
- 局部变量：直接传递指针
- 统一的写回机制

**优势**:
- 语义正确
- 实现统一

### 3. 匿名类的零开销
