```zig
fn parseUseStatement(self: *Parser) !Node.Index {
    // use App\Service;
    // use App\Service as S;
    try self.expect(.keyword_use);
    const name = try self.parseQualifiedName();
    
    var alias: ?StringId = null;
    if (self.match(.keyword_as)) {
        alias = try self.parseIdentifier();
    }
    
    try self.expect(.semicolon);
    
    return try self.addNode(.{
        .tag = .use_stmt,
        .data = .{ .use_stmt = .{ 
            .namespace = name, 
            .alias = alias,
            .use_type = 0  // 0=class, 1=function, 2=const
        } },
    });
}
```

3. **限定名解析**
```zig
fn parseQualifiedName(self: *Parser) !StringId {
    // App\Utils\Helper
    var parts = std.ArrayList([]const u8).init(self.allocator);
    defer parts.deinit();
    
    while (true) {
        const part = try self.parseIdentifier();
        try parts.append(self.getString(part));
        
        if (!self.match(.backslash)) break;
    }
    
    const full_name = try std.mem.join(self.allocator, "\\", parts.items);
    return try self.internString(full_name);
}
```

## 💡 技术要点

### 1. 命名空间的完整流程

```
Parser                    IR Generator              Codegen
  ↓                           ↓                        ↓
namespace_stmt     →    current_namespace    →    完整类名注册
  ↓                           ↓                        ↓
use_stmt           →    namespace_aliases    →    类名解析
  ↓                           ↓                        ↓
class_decl         →    getFullClassName()   →    "App\\Utils\\Helper"
```

### 2. 类名解析优先级

```
1. 完全限定名（\App\Service）→ 去掉前导\
2. 别名（S）→ 查namespace_aliases
3. 导入（Service）→ 查namespace_imports
4. 相对名（Helper）→ 加current_namespace
5. 全局名 → 直接使用
```

### 3. 变量名vs注册名

```zig
// 变量名：短名（用于Zig代码）
const Helper_meta = ...

// 注册名：完整名（用于Runtime查找）
ClassMeta.init(allocator, "App\\Utils\\Helper")
```
